"""Run from the addon root after final tests and disposable environment cleanup."""
import ast
import hashlib
import json
import re
import zipfile
from datetime import datetime, timezone
from pathlib import Path
import xml.etree.ElementTree as ET

root = Path.cwd()
started = datetime.now(timezone.utc).isoformat(timespec='seconds')
report = json.loads((root / 'TEST_REPORT.json').read_text(encoding='utf-8'))
checks = []


def check(label, condition=True):
    assert condition, label
    checks.append(label)


def latest(scope):
    return [r for r in report['history'] if r.get('prompt_id') == 'PR-05'
            and r.get('verification_scope') == scope][-1]


manifest = ast.literal_eval((root / '__manifest__.py').read_text(encoding='utf-8'))
sources = [p for folder in ('models', 'tests', 'scripts') for p in (root / folder).rglob('*.py')]
for source in sources:
    ast.parse(source.read_text(encoding='utf-8-sig'))
check(f'Python syntax: {len(sources)} files')
for path in manifest['data']:
    ET.parse(root / path)
check('All manifest XML data parses')
check('Community dependencies/version retained', manifest['depends'] == ['base', 'product']
      and manifest['version'] == '20.0.1.0.0')
check('Installed parser dependencies declared', manifest['external_dependencies']['python'] == ['openpyxl', 'lxml'])
with zipfile.ZipFile(root / 'evidence/pr04-source-baseline.zip') as archive:
    prior = json.loads((root / 'evidence/pr04-verified-source-sha256.json').read_text(encoding='utf-8'))
    check('Archived PR04 matches every verified source hash', all(p in archive.namelist()
        and hashlib.sha256(archive.read(p)).hexdigest() == h for p, h in prior.items()))
current = json.loads((root / 'evidence/pr05-verified-source-sha256.json').read_text(encoding='utf-8'))
check('Frozen final source hashes match', all(hashlib.sha256((root / p).read_bytes()).hexdigest() == h
    for p, h in current.items()))
fresh = latest('PR05_FRESH')
update = latest('PR05_UPGRADE_VERIFY')
targeted = latest('PR05_TARGETED')
for label, record, count in [('Final fresh install', fresh, 88),
                             ('Final PR04 database update', update, 88),
                             ('Focused intake/security', targeted, 50)]:
    content = (root / record['evidence_reference']).read_text(encoding='utf-8')
    check(label + ': summary/count/status agree', record['result_status'] == 'PASS'
          and record['passed'] == count and record['failed'] == 0
          and f'0 failed, 0 error(s) of {count} tests' in content)
    for cls, n in [('TestImportParser', 11), ('TestImportIntake', 13),
                   ('TestImportSecurity', 7), ('TestImportDownload', 1)]:
        check(label + ': ' + cls + ' selected', len([name for name in record['test_names']
            if name.startswith(cls + '.')]) == n)
probe = latest('PR04_TO_PR05_PRESERVATION')
check('Final actual PR04 preservation probe PASS', probe['result_status'] == 'PASS'
      and 'PR05_PROBE=PASS' in (root / probe['evidence_reference']).read_text(encoding='utf-8'))
concurrency = latest('PR05_CONCURRENCY')
content = (root / concurrency['evidence_reference']).read_text(encoding='utf-8')
check('Two-session extraction and cancellation probes PASS', concurrency['result_status'] == 'PASS'
      and all(marker in content for marker in ['PR05_CONCURRENT_EXTRACT=PASS', 'PR05_CONCURRENT_CANCEL=PASS']))
check('Earlier failures retained', any(r.get('prompt_id') == 'PR-05'
      and r.get('result_status') == 'FAIL' for r in report['history']))
check('Disposable environment removed', not any(root.glob('.pr05*')))
check('Clean shutdown log retained', 'database system is shut down' in
      (root / 'evidence/pr05-postgres-final.log').read_text(encoding='utf-8'))
check('Import architecture and handoff documented', (root / 'docs/IMPORT_ARCHITECTURE.md').is_file())
completed = datetime.now(timezone.utc).isoformat(timespec='seconds')
static_ref = 'evidence/DEV-03-pr05-static-validation.txt'
(root / static_ref).write_text('PR05 QUICK static verification\nStarted: ' + started
    + '\nCompleted: ' + completed + '\n' + '\n'.join('PASS ' + c for c in checks) + '\n', encoding='utf-8')

    'performance_500_csv': dict(status='PASS', row_count=500, seconds=float(timing[-1]) if timing else None,
        reason='One synthetic local intake fixture with exact count/idempotency; not a general throughput claim.'),
    'performance_other_sizes_and_peak_memory': dict(status='NOT_MEASURED', reason='10/100/2000/10000-row '
        'benchmark matrix and peak memory not measured; supported limits do not imply measured load coverage.'),
    'financial_mapping_approval_application': dict(status='NOT_RUN', reason='Deferred to PR06 and later prompts.')}

    'CSV encoding/delimiter are explicit selections; no automatic dialect guessing or business column mapping.',
